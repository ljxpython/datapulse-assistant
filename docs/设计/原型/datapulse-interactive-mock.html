🔍 经过 AST 只读沙箱审定的 SQL 语句 

SELECT 
  t.team_name,
  SUM(s.amount) AS total_gmv,
  ROUND((SUM(s.amount) - LAG(SUM(s.amount)) OVER (PARTITION BY t.team_name ORDER BY s.month)) 
        / LAG(SUM(s.amount)) OVER (PARTITION BY t.team_name ORDER BY s.month) * 100, 2) AS mom_growth
FROM sales_records s
JOIN teams t ON s.team_id = t.id
WHERE s.order_date >= date('now', '-6 month')
GROUP BY t.team_name
ORDER BY total_gmv DESC
LIMIT 500; -- [AST 沙箱自动注入截断]
